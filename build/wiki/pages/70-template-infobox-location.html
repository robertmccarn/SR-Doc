Description[]

Syntax[]

{{infobox location
 | title         = 
 | image         = [e.g. "Example.jpg"]
 | imagewidth    = [e.g. "150"] [default: 210 pixels]
 | imagecaption  = 
 | map           = [e.g. "Example.jpg"]
 | mapwidth      = [e.g. "150"] [default: 210 pixels]
 | mapcaption    = 
 | type          = 
 | level         = 
 | location      = 
 | inhabitants   = 
}}
